FROM delivery_order d
JOIN zone z ON z.zone_id = d.dropoff_zone_id
WHERE d.created_at >= now() - interval '7 days'
  AND d.status = 'DELIVERED'
GROUP BY z.zone_name
ORDER BY revenue DESC;
